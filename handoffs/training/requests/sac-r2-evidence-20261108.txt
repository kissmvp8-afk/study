from pathlib import Path
import hashlib
import json
import tempfile
import zipfile

source = Path('/root/rivermind-data/tita-check/sac-r2-20261108/export-001/SAC_R2_Full_Evidence.zip')
expected_size = 68043449
expected_sha = 'c68d3d958f410d6012007e8801d35555f5e075119486794448ef3cfd19c8a456'
assert source.stat().st_size == expected_size, '原始ZIP字节数不一致，停止'
digest = hashlib.sha256()
with source.open('rb') as file:
    for block in iter(lambda: file.read(1024 * 1024), b''):
        digest.update(block)
assert digest.hexdigest() == expected_sha, '原始ZIP哈希不一致，停止'

output = Path(tempfile.mkdtemp(prefix='readable-r2-', dir=str(source.parent)))
parts = []
with source.open('rb') as file:
    index = 0
    while True:
        data = file.read(25 * 1024 * 1024)
        if not data:
            break
        member = f'{source.name}.part-{index:02d}'
        info = dict(original_file=source.name, original_bytes=expected_size,
                    original_sha256=expected_sha, part_index=index,
                    part_member=member, part_bytes=len(data),
                    part_sha256=hashlib.sha256(data).hexdigest())
        target = output / f'SAC_R2_Evidence_Part{index + 1:02d}.zip'
        with zipfile.ZipFile(target, 'x', compression=zipfile.ZIP_STORED) as archive:
            archive.writestr(member, data)
            archive.writestr('PART_INFO.json', json.dumps(info, ensure_ascii=False, indent=2) + '\n')
        assert target.stat().st_size < 32 * 1024 * 1024
        with zipfile.ZipFile(target) as archive:
            assert archive.testzip() is None
        info.update(archive_file=target.name, archive_bytes=target.stat().st_size,
                    archive_sha256=hashlib.sha256(target.read_bytes()).hexdigest())
        parts.append(info)
        index += 1
assert len(parts) == 3 and sum(item['part_bytes'] for item in parts) == expected_size
(output / 'PARTS_MANIFEST.json').write_text(json.dumps(parts, ensure_ascii=False, indent=2) + '\n')
print(output)
for item in parts:
    print(item['archive_file'], item['archive_bytes'], item['archive_sha256'])
